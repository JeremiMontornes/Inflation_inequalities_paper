% 2
\begin{frame}{What We Do}
\begin{itemize}
  \item Inflation inequality across households increases when relative-price shocks interact with heterogeneous household baskets.
  \item We construct monthly indices by income, age, and residence for all 20 euro-area countries, calibrated to official HICP weights.
  \item We simulate the effects of temporary VAT cuts, gas and electricity price measures, and fuel rebates on average inflation and inflation inequality.
\end{itemize}
\vspace{0.5em}
\textbf{Our contribution}
\begin{enumerate}
  \item A comprehensive framework to assess inflation inequality.
  \item We connect household inflation exposure to price policies, quantifying their effects on both average inflation and income-group gaps.
\end{enumerate}
\end{frame}

% 3
\begin{frame}{Inflation Inequality and Policy Effects, 2021--2023}
\begin{itemize}
  \item Euro-area Q1--Q5 inflation gap rose to \textbf{0.9 point}, from 0.1 point in 2021, and returned to zero in 2023.
  \item Group averages conceal substantial within-category heterogeneity: households in the same income quintile can face markedly different inflation rates.
  \item Household energy contributed \textbf{1.6 points} to the 2022 Q1--Q5 gap; fuel, transport, and other products offset most of this exposure.
  \item Household price policies lowered annual inflation by \textbf{1.3 points in 2022} and \textbf{0.6 point in 2023}, while narrowing the Q1--Q5 gap by \textbf{0.4} and \textbf{0.5 point}.
\end{itemize}
\end{frame}

% 4
\begin{frame}{Linking Official Prices to Household Baskets}
\begin{itemize}
  \item We link monthly Eurostat HICP prices and official item weights to HBS expenditure shares through a harmonized COICOP bridge.
  \item Public HBS data provide consumption baskets by income, age, and residence; microdata add household expenditures, disposable income, and demographics.
  \item We use RAS to calculate expenditure weights that simultaneously satisfy two margins: the expenditure share of each household category and the official HICP weight of each product.
\end{itemize}
\vspace{0.5em}
\begin{itemize}
  \item Interpretation: baskets are held fixed between HBS waves, so the baseline abstracts from substitution and within-group heterogeneity (behavioral responses are not taken into account).
\end{itemize}
\end{frame}

% 11
\begin{frame}{Three Instruments Lowered Retail Energy Prices}
\small
The counterfactual covers \textbf{100 price-policy measures} (prices held below market levels or tax cuts) across all 20 euro-area countries, costing \textbf{EUR 268.9 billion}; \textbf{88\%} of the cost was untargeted.
\par\vspace{0.5em}
\begin{itemize}
  \item \textbf{Price caps and tariff shields} limited household electricity and gas prices.\\
  \textit{France: gas tariffs were frozen and the 2022 electricity increase was capped at 4\%.}
  \item \textbf{Temporary VAT and excise-tax reductions} lowered consumer energy prices.
  \item \textbf{Fuel rebates and regulated discounts} reduced prices at the pump.\\
  \textit{France: the fuel rebate rose from EUR 0.18 to EUR 0.30 per litre in autumn 2022.}
\end{itemize}
\source{National policy sources; Sgaravatti et al.; authors' calculations.}
\end{frame}

% 12
\begin{frame}{Defining the Price-Policy Counterfactual}
\begin{itemize}
  \item The counterfactual removes these price-based interventions component by component while retaining observed prices for unaffected products.
\end{itemize}
\vspace{0.4em}
\textbf{Example: reversing a temporary VAT cut}
\[
\widetilde p_{j,t}
=
p_{j,t}
\frac{1+\tau^{0}_{j,t}}{1+\tau^{1}_{j,t}},
\qquad
\tau^{0}_{j,t}=\text{VAT rate},\quad
\tau^{1}_{j,t}=\text{temporary VAT rate}.
\]
\centering
\textit{Belgium: cutting VAT from \textbf{21\%} to \textbf{6\%} implies}
\[
\widetilde p_{j,t}=p_{j,t}\frac{1.21}{1.06}\simeq 1.142\,p_{j,t}.
\]
{\small
$j=\text{electricity}$ and $t=$ March 2022--December 2023.
}
\source{National policy sources, Sgaravatti et al.; authors' calculations.}
\end{frame}

\begin{frame}{Household-Level Price Indices}
\begin{itemize}
  \item We treat each household as an elementary group. Its individual weight $\omega^{\mathrm{HBS}}_{j,i,2020}$ combines its expenditure on item $j$ with the official HICP weight for that item.
\end{itemize}
\vspace{0.5em}
\[
P_{i,y,m}
=
\sum_{j \in \mathcal{J}_{i,2020,y}}
\omega^{\mathrm{HBS}}_{j,i,2020}
\frac{p_{j,y,m}}{p_{j,y,0}}.
\]
\vspace{0.5em}
\begin{itemize}
  \item Individual indices measure inflation heterogeneity within income, age, and residence categories, which broad group averages conceal.
  \item This within-category dispersion was substantial during the 2021--2023 inflation surge.
\end{itemize}
\end{frame}

% 17
\begin{frame}{Price Policies Reduced Average Inflation and Inflation Inequality}
\begin{itemize}
  \item The inflation shock affected households very differently, even at a given income level: most inflation dispersion occurs \textbf{within} income groups.
  \item Q1--Q5 inflation inequality rose from \textbf{0.1 point in 2021} to \textbf{0.9 point in 2022}, then returned to zero in 2023; national experiences remained far more dispersed.
  \item Household energy contributed \textbf{1.6 points} to the 2022 Q1--Q5 gap, while fuel, transport, and other consumption categories offset it.
  \item Price policies reduced annual inflation by \textbf{1.3 points in 2022} and \textbf{0.6 point in 2023}; they reduced inequality by \textbf{0.4} and \textbf{0.5 point}.
  \item Administered household-energy prices generated most of the distributional cushioning; fuel support was mildly regressive in 2022.
  \item Our policy counterfactuals quantify effects on consumer prices and inflation gaps, rather than a complete welfare assessment.
\end{itemize}
\end{frame}
