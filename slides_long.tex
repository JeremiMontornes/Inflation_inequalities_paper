\begin{frame}{Household Inflation Exposure in the Euro Area}
\begin{itemize}
\item Our question: How have recent energy shocks and the fiscal response shaped household- and group-level inflation
 across the euro area?

\item We construct:
\begin{itemize}
\item Monthly household-level price indices for approximately 150,000 households.
\item Monthly group-level price indices by income, age, and residence area.
\end{itemize}
\item Counterfactual price indices to quantify how fiscal measures changed average inflation during the energy crisis.

\end{itemize}
\end{frame}

\begin{frame}{Main Findings}
\begin{itemize}
\item The annual Q1--Q5 inflation gap reached \textbf{1.2 percentage points in 2022}, with large cross-country differences.
\item Price policies lowered inflation by \textbf{1.5 percentage points in 2022} and reduced the Q1--Q5 inflation gap by \textbf{0.6 percentage points}.
\item Household-level inflation displays substantial dispersion within income quintiles.
\item Results are based on the Harmonised Index of Consumer Prices (HICP) and Household Budget Surveys (HBS).
\end{itemize}

\end{frame}

% Evidence: main.tex, data and methodology
\begin{frame}{Linking HICP Prices to Household Budget Surveys}
\begin{itemize}
\item \textbf{Prices:} monthly HICP components and annual official product weights.
\item \textbf{Consumption Baskets:} fixed 2020 HBS household expenditure; group baskets are built from microdata where available.
\item \textbf{Coverage:} group-level indices for 20 countries; household-level indices for 18 countries.
\item \textbf{Matching:} remove actual and imputed rents, then calibrate matched three-digit COICOP products.
\end{itemize}
\end{frame}

% Evidence: main.tex, group price indices
\begin{frame}{Calibrated Baskets Produce Chained Inflation Indices}

\par\vspace{.5em}
{\textbf{Inflation inequality R package:} download, calculate, visualize\par
\href{https://github.com/JeremiMontornes/inflationinequality}{\textcolor{Teal}{github.com/JeremiMontornes/inflationinequality}}\par}
\small
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\textbf{Calibrate expenditure weights}
\par\medskip
RAS adjusts the product-by-group matrix $M$ to match group expenditure shares $s_{g,y}$ and HICP product shares:
\[
\sum_j M_{j,g,y}=s_{g,y},\qquad
\sum_g M_{j,g,y}=\bar\omega^{\mathrm{HICP}}_{j,y}.
\]
\[
\omega_{j,g,y}=\frac{M_{j,g,y}}{s_{g,y}},\qquad
\sum_g s_{g,y}\omega_{j,g,y}=\bar\omega^{\mathrm{HICP}}_{j,y}.
\]
Use expenditure shares, not population shares; matched products exclude rents.
\end{column}
\begin{column}{.48\textwidth}
\textbf{Construct and chain price indices}
\par\medskip
Compare product prices with the preceding December:
\[
P_{g,y,m}=\sum_j\omega_{j,g,y}
\frac{p_{j,y,m}}{p_{j,y-1,12}}.
\]
\[
I_{g,y,m}=I_{g,y-1,12}P_{g,y,m}.
\]
Euro-area indices combine country series using annual HICP country weights.
\end{column}
\end{columns}

\end{frame}

\begin{frame}{Conclusion}
\small
\begin{itemize}
\setlength{\itemsep}{.5em}
\item \textbf{Disaggregated inflation indicators.} HICP prices and household baskets reveal heterogeneous inflation exposure across and within groups.
\item \textbf{Unconventional fiscal policy.} UFP reduced euro-area inflation and compressed inflation inequality. The counterfactual identifies direct consumer-price effects; production-network effects are analyzed separately for import-price shocks (Gautier and Montorn\`es, 2026; working paper forthcoming).
\item \textbf{Effectiveness was state-dependent.} UFP was effective, but monetary policy and favorable circumstances also played a role: the energy-price shock was temporary, and the euro-area economy was not especially overheated, unlike the U.S. economy.
\item \textbf{Any questions?}
\end{itemize}
\end{frame}

\begin{frame}{Household Microdata Show Unequal Inflation Exposure}
\begin{itemize}
\item \textbf{Household-specific baskets:} 152,613 households across
18 countries (excluding Italy and Portugal).
\item \textbf{Household-specific inflation:} non-rent expenditure shares
are matched to country-level HICP price changes.
\item \textbf{What the measure captures:} inflation differences
arising from households' consumption baskets. Common HICP prices do not capture differences
in prices paid for the same goods, a major source of household inflation
dispersion in Kaplan and Schulhofer-Wohl (2017).
\end{itemize}
\end{frame}

\begin{frame}{Group and Household Indices Both Exclude Rents}
\begin{itemize}
\item Exclude actual and imputed rents and corresponding HICP rent components before matching and aggregation.
\item Reconstruct parent housing categories without rents and normalize weights over matched non-rent products.
\item Both group and household indices use RAS calibration to annual HICP product margins, with matched non-rent coverage.
\end{itemize}
\par\vspace{.5em}
Rent sensitivity and official all-items validation retain their explicitly stated reference coverage.
\end{frame}

\begin{frame}{RAS Calibration and the Micro--Macro Link}
\small
$a_{ic}$: survey weight; $E_{ic}$: matched non-rent expenditure; $L_{ic,y,m}$: December-relative household price link.
\[
L_{c,y,m}=\frac{\sum_i a_{ic}E_{ic}L_{ic,y,m}}{\sum_i a_{ic}E_{ic}},
\qquad L_{y,m}=\sum_c q_{c,y}L_{c,y,m}.
\]
\[
I_{y,m}=I_{y-1,12}L_{y,m}.
\]
$q_{c,y}$: annual HICP country weights, adjusted for rents and normalized over included countries.
\par\medskip
\textbf{Aggregate links before chaining.} Calculate inflation from the resulting aggregate index.
\par\medskip
The country-weighted average of national household means is a separate distributional statistic.
\end{frame}

\begin{frame}{Inflation Burden Scales Exposure by Household Resources}
For household $i$ in country $c$, the raw exposure measure is
\[
\mathcal B^{\mathrm{raw}}_{ic,y}=\bar\pi_{ic,y}\frac{X_{ic}}{Y_{ic}}.
\]
\begin{itemize}
\item $\bar\pi$: annual household inflation (\%); $X$: matched non-rent expenditure; $Y$: net income, both from the 2020 HBS.
\item Winsorize expenditure-to-income ratios at the national survey-weighted 99th percentile, then aggregate household burdens.
\item Non-rent inflation is scaled by matched non-rent expenditure relative to income: this measures exposure relative to income, rather than an exact welfare loss.
\end{itemize}
\end{frame}

\begin{frame}{Counterfactual Effects Use Two Identification Methods}
\small
For product $j$ and month $t$, multiply the observed HICP index by its counterfactual-to-observed retail-price ratio:
\[
\widetilde p_{j,t}=p_{j,t}\frac{\widetilde r_{j,t}}{r_{j,t}}.
\]
\begin{itemize}
\item \textbf{Source-determined effects (102 entries):} statutory schedules, regulated tariffs, or official administrative price and bill series determine the effect.
\item \textbf{Model-imputed effects (90 entries):} an aggregate-equivalent wedge, representative bill or contract, pass-through assumption, or model proxy determines the effect.
\item Table D.1 covers \textbf{192 measure--product entries} from \textbf{86 distinct policies}; its euro-area row sums the 20 country rows.
\end{itemize}
\end{frame}

\qquad

+\mathbf w_g^{\mathsf T}\mathbf C\Delta\mathbf p^k.

\begin{frame}{Question 1: What Generates Inflation Inequality?}
\textbf{Which combination can generate inflation inequality through consumption baskets?}
\par\vspace{.6em}
{\small\textit{Select one answer.}\par}
\vspace{.6em}
\begin{enumerate}
\item[A.] Different consumption baskets and different price changes across products.
\item[B.] Different consumption baskets when all product prices rise at the same rate.
\item[C.] Differences in wage indexation alone.
\item[D.] Differences in savings and debt alone.
\end{enumerate}
\end{frame}

% Q2 answer: A. The HICP--HBS indicators capture basket heterogeneity across
% COICOP categories, but not product varieties or transaction prices within categories.
\begin{frame}{Question 2: What Do Our Inflation Indicators Capture?}
\textbf{Which source of household inflation heterogeneity is captured by our HICP--HBS indicators?}
\par\vspace{.6em}
{\small\textit{Select one answer.}\par}
\vspace{.6em}
\begin{enumerate}
\item[A.] Differences in expenditure shares across COICOP categories, across groups and households.
\item[B.] Differences in product varieties, brands and pack sizes within the same category.
\item[C.] Differences in retailers, promotions and purchase dates for identical products.
\end{enumerate}
\end{frame}

% Q3 answers: B, C and D. Monetary policy can look through the first-round
% effect of a temporary supply shock, but it must prevent second-round effects.
\begin{frame}{Question 3: Policy Responses to a Supply Shock}
\textbf{Which statements correctly describe the policy response to an adverse supply shock?}
\par\vspace{.6em}
{\small\textit{Several answers are possible. Select all that apply.}\par}
\vspace{.6em}
\begin{enumerate}
\item[A.] Monetary policy should never respond because inflation originates from a supply shock.
\item[B.] Monetary policy should prevent second-round effects and a de-anchoring of inflation expectations.
\item[C.] Conventional fiscal policy can support the policy mix by tightening aggregate demand.
\item[D.] Temporary, targeted unconventional fiscal measures can support monetary policy but cannot replace it.
\end{enumerate}
\end{frame}

\begin{frame}{The Answers Follow Three Core Mechanisms}
\large
\textcolor{Teal}{\textbf{Question 1 --- A}}\par
Different baskets generate different inflation rates only when relative price changes differ across products.

\par\vspace{1em}
\textcolor{Teal}{\textbf{Question 2 --- A}}\par
Our indicators capture expenditure-share differences across COICOP categories, but not varieties or transaction prices within categories.

\par\vspace{1em}
\textcolor{Teal}{\textbf{Question 3 --- B, C and D}}\par
Monetary policy must prevent second-round effects, while conventional fiscal tightening and temporary, targeted price support can complement that response.
\end{frame}
